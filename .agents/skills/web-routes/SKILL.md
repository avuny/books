# Web routes (TanStack Router, file-based)

Routes are **thin glue**: read params → create navigation callbacks → render one feature component. No fetching, no translation, no styling logic.

## Where files go (mirror of the real `organization` tree)

```
routes/organization/
├── route.tsx                    layout: auth guard  → "/organization"
├── index.tsx                    page                → "/organization/"
├── create.tsx                   page                → "/organization/create"
└── $orgId/
    ├── route.tsx                layout: dashboard   → "/organization/$orgId"
    ├── index.tsx                page                → "/organization/$orgId/"
    ├── role.tsx                 redirect → roles    → "/organization/$orgId/role"
    ├── users.tsx                leaf page           → "/organization/$orgId/users"
    ├── roles/
    │   ├── index.tsx            list                → "/organization/$orgId/roles/"
    │   └── $roleId.tsx          details             → "/organization/$orgId/roles/$roleId"
    └── settings/
        ├── route.tsx            nested layout       → "/organization/$orgId/settings"
        └── index.tsx            page                → "/organization/$orgId/settings/"
```

**The string in `createFileRoute` must equal the file location.** `index.tsx` → trailing `/`. Never edit `routeTree.gen.ts` (generated by the router plugin / dev server).

## Templates for a new `projects` area

### List page — `routes/organization/$orgId/projects/index.tsx`

```tsx
import { createFileRoute, useNavigate } from "@tanstack/react-router"

import { ProjectList } from "@/features/project-module"

export const Route = createFileRoute("/organization/$orgId/projects/")({
  component: ProjectsPage,
})

function ProjectsPage() {
  const { orgId } = Route.useParams()
  const navigate = useNavigate()

  return (
    <ProjectList
      orgId={orgId}
      onViewProject={(projectId) =>
        navigate({
          to: "/organization/$orgId/projects/$projectId",
          params: { orgId, projectId },
        })
      }
    />
  )
}
```

### Details page — `routes/organization/$orgId/projects/$projectId.tsx`

```tsx
import { createFileRoute, useNavigate } from "@tanstack/react-router"

import { ProjectDetails } from "@/features/project-module"

export const Route = createFileRoute("/organization/$orgId/projects/$projectId")({
  component: ProjectDetailsPage,
})

function ProjectDetailsPage() {
  const { orgId, projectId } = Route.useParams()
  const navigate = useNavigate()

  return (
    <ProjectDetails
      orgId={orgId}
      projectId={projectId}
      onBack={() => navigate({ to: "/organization/$orgId/projects", params: { orgId } })}
    />
  )
}
```

### Redirect alias — `…/$orgId/project.tsx` (singular → plural)

```tsx
import { createFileRoute, redirect } from "@tanstack/react-router"

export const Route = createFileRoute("/organization/$orgId/project")({
  beforeLoad: ({ params }) => {
    throw redirect({ to: "/organization/$orgId/projects", params: { orgId: params.orgId } })
  },
})
```

### Layout — `route.tsx`

```tsx
import { Outlet, createFileRoute } from "@tanstack/react-router"
import { SomeLayout } from "@/features/some-feature"

export const Route = createFileRoute("/organization/$orgId/settings")({
  component: SettingsRoute,
})

function SettingsRoute() {
  const { orgId } = Route.useParams()
  return (
    <SomeLayout orgId={orgId}>
      <Outlet />
    </SomeLayout>
  )
}
```

Layout opt-out (settings uses its own layout): in `$orgId/route.tsx` check `useLocation().pathname.startsWith("/organization/${orgId}/settings")` and return a bare `<Outlet />`.

### Auth guard — only in the top layout (`organization/route.tsx`)

```tsx
function OrganizationLayout() {
  const { data, isLoading } = useAuthenticatedHandler()
  if (isLoading) return <LoadingPage />
  if (!data?.success) return <Navigate to={ROUTES.auth.index} replace />
  return <div className="flex h-screen flex-col bg-muted"><Outlet /></div>
}
```

Child routes never repeat the check. Use `ROUTES` from `@/features/routes` instead of string literals for well-known app paths.

### Search params — `validateSearch`

```tsx
type ProjectsSearch = { archived?: boolean }

export const Route = createFileRoute("/organization/$orgId/projects/")({
  validateSearch: (search: Record<string, unknown>): ProjectsSearch => ({
    archived: search.archived === true || search.archived === "true",
  }),
  component: ProjectsPage,
})
// read: const { archived } = Route.useSearch()
// write: navigate({ to: "/organization", search: { all: true } })
```

### Page with the top navbar (standalone pages like create/index)

```tsx
<>
  <Navbar
    start={<><HomeButton /><DashboardButton /></>}
    end={<><UserButton /><ModeSwitcherBtn /><LanguageSwitcherButton /></>}
  />
  <div className="flex-1 overflow-auto p-6 md:p-8">
    <CreateOrganization onSuccess={(org) => { void navigate({ to: "/organization/$orgId", params: { orgId: org.id } }) }} />
  </div>
</>
```

## Rules of thumb

- Typed navigation: `to` is the route's path pattern, `params` object carries the `$…` values.
- Fire-and-forget navigations in callbacks: `void navigate(...)`; returning the promise (`onViewProject={(id) => navigate(...)}`) is fine for callbacks typed `=> void`.
- Redirect effect on mount (`index.tsx` jumping to the selected org): `navigate({ …, replace: true })` inside `useEffect`, and `return null` while redirecting.
- A placeholder page still uses `t()`; don't ship hard-coded English like `users.tsx` does.

## Checklist

- [ ] File only: params, navigate callbacks, one feature component
- [ ] Back/after-success navigation implemented here, not in the feature
- [ ] Legacy/singular URL? add a redirect leaf
